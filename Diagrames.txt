###################################### Classe #####################################

@startuml

Class Grille{
Longueur : int
Largeur : int
Perdu : bool
---
placer_drapeau(case)
creuser(case)
voisins(case)
}

Class Case {
Position : (int, int)
Bombe : bool
Etat : etat (drapeau, découvert, cachée)
---
placer_drapeau()
découvrir()
nombre_voisins()
}


Grille "1" *-- Case

@enduml

###################################################################################
##################################### Séquence #################################### 
###################################################################################

@startuml

skinparam actorStyle awesome
Actor Utilisateur
hide footbox

==Initialisation==

Utilisateur -> Jeu : Commencer partie
Jeu --> Utilisateur : Grille vide

==Partie==

alt Révéler case
   Utilisateur -> Jeu : Creuser la case
   Jeu -> Case : Creuser
   alt Pas de bombe
      Case --> Jeu : Case révélée
      Jeu -> Case : Bombe autour de la case
   
   else Bombe présente
      Case --> Jeu : Explosion
      Jeu --> Utilisateur : Fin de partie, défaite
   end
      

else Placer drapeau
   Utilisateur -> Jeu : Placer un drapeau sur la case
   Jeu -> Case : Placer un drapeau

end

==Fin==

Jeu -> Jeu : Plus de mines
Jeu --> Utilisateur : Fin de partie, victoire


@enduml

################################ Cas d'utilisation ################################

  usecase "Creuser une case" as UC3

@enduml
